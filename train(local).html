inputs = keras.Input(shape=(img_size, img_size, 1))

data_augmentation = keras.Sequential([
    layers.RandomRotation(0.1),
    layers.RandomZoom(0.1),
    layers.RandomTranslation(0.1, 0.1),
])
# Apply augmentation only during training
x = data_augmentation(inputs)
x = layers.Rescaling(1./255)(x)

# Helper function to create a Conv Block with BatchNormalization
def conv_block(x, filters):
    x = layers.Conv2D(filters, (3, 3), padding='same', use_bias=False)(x)
    x = layers.BatchNormalization()(x)
    x = layers.Activation('relu')(x)
    x = layers.MaxPooling2D((2, 2))(x)
    return x

# Deeper Architecture
x = conv_block(x, 32)
x = conv_block(x, 64)
x = conv_block(x, 128)
x = conv_block(x, 256)

x = layers.Flatten()(x)

# Dense Block
x = layers.Dense(256, use_bias=False)(x)
x = layers.BatchNormalization()(x)
x = layers.Activation('relu')(x)
x = layers.Dropout(0.5)(x)

outputs = layers.Dense(num_classes, activation='softmax')(x)

model = keras.Model(inputs=inputs, outputs=outputs)

model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

model.summary()

y_true_all = []
y_pred_all = []

for images, labels in val_ds:
    # Get predictions
    preds = model.predict(images, verbose=0)
    
    # Convert predictions to class indices (0-61)
    pred_indices = np.argmax(preds, axis=1)
    
    # Handle True Labels
    # If labels are one-hot (from the categorical update), convert to indices
    if len(labels.shape) > 1 and labels.shape[1] > 1:
        true_indices = np.argmax(labels.numpy(), axis=1)
    else:
        # If labels are integers (original setup)
        true_indices = labels.numpy()
        
    y_true_all.extend(true_indices)
    y_pred_all.extend(pred_indices)

# 2. Calculate Accuracy Per Class
# Create confusion matrix
cm = confusion_matrix(y_true_all, y_pred_all, labels=list(label_to_char_map.keys()))

# Diagonal = Correct predictions per class
# Sum(axis=1) = Total samples per class
# We add a tiny epsilon to avoid division by zero errors
class_accuracy = cm.diagonal() / (cm.sum(axis=1) + 1e-10)

# 3. Plotting
# Get character labels in order 0-9, A-Z, a-z
labels_text = [label_to_char_map[i] for i in range(len(label_to_char_map))]

plt.figure(figsize=(20, 8)) # Wide figure to fit 62 bars

# Create Bar Chart
# We assign colors: Green for high accuracy, Red for low
colors = ['green' if acc > 0.8 else 'red' for acc in class_accuracy]

bars = plt.bar(labels_text, class_accuracy, color=colors, edgecolor='black', alpha=0.7)

plt.xlabel('Characters', fontsize=14)
plt.ylabel('Accuracy (0.0 - 1.0)', fontsize=14)
plt.title('Prediction Accuracy per Character', fontsize=18)
plt.xticks(rotation=90, fontsize=10) # Vertical text for readability
plt.ylim(0, 1.05) # Y-axis from 0 to 100%

# Add grid for easier reading
plt.grid(axis='y', linestyle='--', alpha=0.5)

# Add a horizontal line for the Global Average
global_avg = np.mean(class_accuracy)
plt.axhline(global_avg, color='blue', linestyle='--', linewidth=2, label=f'Mean Acc: {global_avg:.2f}')
plt.legend()

plt.tight_layout()
plt.show()
